// Factory + Observer class News { constructor(title, content) { this.title = title; this.content = content; } getInfo() { return `[Новина] ${this.title}: ${this.content}`; } } class SportsNews extends News { getInfo() { return `[⚽ Спорт] ${this.title}: ${this.content}`; } } class PoliticsNews extends News { getInfo() { return `[🏛️ Політика] ${this.title}: ${this.content}`; } } class EntertainmentNews extends News { getInfo() { return `[🎬 Розваги] ${this.title}: ${this.content}`; } } // Factory Method class NewsFactory { static create(type, title, content) { switch (type) { case 'sport': return new SportsNews(title, content); case 'politics': return new PoliticsNews(title, content); case 'entertainment': return new EntertainmentNews(title, content); default: return new News(title, content); } } } // Observer class NewsAgency { constructor() { this.subscribers = []; } subscribe(subscriber) { this.subscribers.push(subscriber); } unsubscribe(subscriber) { this.subscribers = this.subscribers.filter(s => s !== subscriber); } publish(news) { console.log("📰 Нова новина надійшла:", news.getInfo()); this.subscribers.forEach(sub => sub.notify(news)); } } class User { constructor(name) { this.name = name; } notify(news) { console.log(`👤 ${this.name} отримав новину: ${news.getInfo()}`); } } // === Приклад використання === const agency = new NewsAgency(); // Створюємо користувачів const user1 = new User("Оля"); const user2 = new User("Андрій"); // Підписка agency.subscribe(user1); agency.subscribe(user2); // Створюємо новину через фабрику const news1 = NewsFactory.create("sport", "Україна виграла!", "Наша збірна перемогла 3:1"); const news2 = NewsFactory.create("politics", "Зміни в уряді", "Новий міністр освіти"); // Публікуємо agency.publish(news1); agency.publish(news2); // Відписка та ще одна новина agency.unsubscribe(user2); const news3 = NewsFactory.create("entertainment", "Прем’єра фільму", "Вийшов новий український блокбастер"); agency.publish(news3);
